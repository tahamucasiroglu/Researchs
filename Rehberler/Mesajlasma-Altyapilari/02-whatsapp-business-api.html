  • Onay akışı: PENDING → APPROVED / REJECTED (sonradan PAUSED, DISABLED olabilir). Sonuç message_template_status_update webhook'u ile gelir. İnceleme genelde dakikalar, en çok ~24 saat.
  • Sık ret sebepleri: değişken biçim hatası (art arda değişken, değişkenle başlama/bitme), örnek değer eksik, hassas kimlik bilgisi isteme, mevcut şablonun kopyası.
  • Kategori değişimi: Meta uygun olmayan Utility şablonu kendiliğinden Marketing'e çevirir (9 Nis 2025'ten beri allow_category_change desteklenmiyor). Kategori fiyatı belirlediği için maliyete yansır.
  • Kalite: şablon başına GREEN / YELLOW / RED; düşük kalitede gönderim yavaşlatılır (pacing) ya da şablon duraklatılır (3 sa → 6 sa → devre dışı).
  • Düğmeler: QUICK_REPLY, URL, PHONE_NUMBER, COPY_CODE, OTP, FLOW, CATALOG, MPM, VOICE_CALL…; şablon başına en çok 10.

§ 8Webhook

İstekNe yapmalısın
GET /webhook?hub.mode=subscribe&hub.verify_token=…&hub.challenge=1158201444Jeton senin belirlediğinle eşleşiyorsa hub.challenge değerini düz metin + 200 dön; değilse 403.
POST /webhook + X-Hub-Signature-256: sha256=<hex>Ham gövde üzerinde uygulama gizli anahtarıyla HMAC-SHA256 hesapla, sabit zamanlı karşılaştır. Hemen 200 dön, işi kuyruğa at.
  • Adres herkese açık HTTPS; geçerli TLS; kendinden imzalı sertifika desteklenmez. Geliştirmede tünel (ngrok, Cloudflare Tunnel) yeterli. RESMÎ
  • 200 dışındaki her cevap tekrar denemeye yol açar. Süre: Graph genel belgesi "36 saat", WhatsApp webhook genel bakış sayfası "7 güne kadar, azalan sıklıkla" diyor ÇELİŞKİ. Her iki durumda: aynı olay birden çok gelebilir → wamid + durum ile tekilleştir.
  • Gövde 3 MB'a kadar. İsteğe bağlı mTLS: Meta CA dosyası meta-outbound-api-ca-2025-12.pem.
  • Abone olunacak alanlar (App Dashboard → WhatsApp → Configuration): en az messages; ihtiyaca göre message_template_status_update, account_update, phone_number_quality_update.

§ 9C# örnekleri (.NET, HttpClient + System.Text.Json)

Gizli değerler ortam değişkeninden: WA_TOKEN, WA_PHONE_NUMBER_ID, WA_APP_SECRET, WA_VERIFY_TOKEN. Kodda, depoda ya da günlükte jeton tutma.

İstemci: şablon ve düğmeli mesaj gönder

using System.Net.Http.Headers;
using System.Net.Http.Json;
using System.Text.Json.Nodes;

public sealed class WhatsAppClient
{
    private readonly HttpClient _http;
    private readonly string _phoneNumberId;

    public WhatsAppClient(HttpClient http)
    {
        _http = http;
        _http.BaseAddress = new Uri("https://graph.facebook.com/v26.0/");
        _http.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Bearer",
            Environment.GetEnvironmentVariable("WA_TOKEN") ?? throw new InvalidOperationException("WA_TOKEN yok"));
        _phoneNumberId = Environment.GetEnvironmentVariable("WA_PHONE_NUMBER_ID")
            ?? throw new InvalidOperationException("WA_PHONE_NUMBER_ID yok");
    }

    // Şablon: pencere dışındaki tek yol
    public Task<string> SendTemplateAsync(string to, string name, string lang, params string[] bodyParams) =>
        SendAsync(new JsonObject
        {
            ["messaging_product"] = "whatsapp",
            ["to"] = to,
            ["type"] = "template",
            ["template"] = new JsonObject
            {
                ["name"] = name,
                ["language"] = new JsonObject { ["code"] = lang },
                ["components"] = new JsonArray(new JsonObject
                {
                    ["type"] = "body",
                    ["parameters"] = new JsonArray(bodyParams
                        .Select(p => (JsonNode)new JsonObject { ["type"] = "text", ["text"] = p })
                        .ToArray())
                })
            }
        });

    // Düğmeli mesaj: yalnız 24 saat penceresi içinde; en çok 3 düğme, başlık 20 karakter
    public Task<string> SendButtonsAsync(string to, string body, params (string Id, string Title)[] buttons)
    {
        if (buttons.Length is 0 or > 3) throw new ArgumentException("1-3 düğme");
        return SendAsync(new JsonObject
        {
            ["messaging_product"] = "whatsapp",
            ["to"] = to,
            ["type"] = "interactive",
            ["interactive"] = new JsonObject
            {
                ["type"] = "button",
                ["body"] = new JsonObject { ["text"] = body },
                ["action"] = new JsonObject
                {
                    ["buttons"] = new JsonArray(buttons.Select(b => (JsonNode)new JsonObject
                    {
                        ["type"] = "reply",
                        ["reply"] = new JsonObject { ["id"] = b.Id, ["title"] = b.Title }
                    }).ToArray())
                }
            }
        });
    }

    private async Task<string> SendAsync(JsonObject payload)
    {
        using var res = await _http.PostAsJsonAsync($"{_phoneNumberId}/messages", payload);
        var json = JsonNode.Parse(await res.Content.ReadAsStringAsync());
        if (!res.IsSuccessStatusCode)
            throw new HttpRequestException($"WhatsApp {(int)res.StatusCode}: " +
                $"{json?["error"]?["code"]} {json?["error"]?["message"]}");
        return json!["messages"]![0]!["id"]!.GetValue<string>();   // wamid — sakla
    }
}

Webhook: doğrula + imza denetle + ayrıştır (ASP.NET Core minimal API)

using System.Security.Cryptography;
using System.Text;
using System.Text.Json;

var app = WebApplication.Create(args);
string verifyToken = Environment.GetEnvironmentVariable("WA_VERIFY_TOKEN")!;
byte[] appSecret  = Encoding.UTF8.GetBytes(Environment.GetEnvironmentVariable("WA_APP_SECRET")!);

// 1) Doğrulama (GET)
app.MapGet("/webhook", (HttpRequest r) =>
    r.Query["hub.mode"] == "subscribe" && r.Query["hub.verify_token"] == verifyToken
        ? Results.Text(r.Query["hub.challenge"].ToString())
        : Results.StatusCode(403));

// 2) Olaylar (POST)
app.MapPost("/webhook", async (HttpRequest r) =>
{
    using var ms = new MemoryStream();
    await r.Body.CopyToAsync(ms);                        // ham gövde: imza bunun üzerinde
    byte[] raw = ms.ToArray();

    string header = r.Headers["X-Hub-Signature-256"].ToString();
    if (!header.StartsWith("sha256=")) return Results.Unauthorized();
    byte[] expected = HMACSHA256.HashData(appSecret, raw);
    byte[] given;
    try { given = Convert.FromHexString(header["sha256=".Length..]); }
    catch (FormatException) { return Results.Unauthorized(); }
    if (!CryptographicOperations.FixedTimeEquals(expected, given)) return Results.Unauthorized();

    using var doc = JsonDocument.Parse(raw);
    foreach (var entry in doc.RootElement.GetProperty("entry").EnumerateArray())
    foreach (var change in entry.GetProperty("changes").EnumerateArray())
    {
        if (change.GetProperty("field").GetString() != "messages") continue;
        var value = change.GetProperty("value");

        if (value.TryGetProperty("messages", out var msgs))
            foreach (var m in msgs.EnumerateArray())
            {
                string from = m.GetProperty("from").GetString()!;
                string id   = m.GetProperty("id").GetString()!;      // tekilleştirme anahtarı
                string type = m.GetProperty("type").GetString()!;
                string? content = type switch
                {
                    "text"        => m.GetProperty("text").GetProperty("body").GetString(),
                    "button"      => m.GetProperty("button").GetProperty("payload").GetString(),
                    "interactive" => InteractiveId(m.GetProperty("interactive")),
                    "image" or "document" or "audio" or "video"
                                  => m.GetProperty(type).GetProperty("id").GetString(), // 7 gün içinde indir
                    _             => null                                       // diğer tipleri de kaydet
                };
                // Kuyruğa at: (id, from, type, content) — burada uzun iş yapma
            }

        if (value.TryGetProperty("statuses", out var sts))
            foreach (var s in sts.EnumerateArray())
            {
                string wamid  = s.GetProperty("id").GetString()!;
                string status = s.GetProperty("status").GetString()!;   // sent / delivered / read / failed
                // Giden mesaj kaydını güncelle
            }
    }
    return Results.Ok();                                 // hızlı 200 — yoksa Meta tekrar dener
});

app.Run();

// button_reply / list_reply → seçilen satırın/düğmenin id'si
static string? InteractiveId(JsonElement i) => i.GetProperty("type").GetString() switch
{
    "button_reply" => i.GetProperty("button_reply").GetProperty("id").GetString(),
    "list_reply"   => i.GetProperty("list_reply").GetProperty("id").GetString(),
    _              => null    // nfm_reply (Flow) vb.
};

Kendi işletmen için (doğrudan Cloud API)
