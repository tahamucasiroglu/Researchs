Şu URL → bunu yapar. TurkeySMS v4 örneğiyle uç noktalar, JSON, webhook imzası, C# kodu; ardından BTK / İYS kuralları, fiyatlar ve karar tablosu.

TurkeySMS API 4.0.0 (OpenAPI 3.1) · Doğrulama: 12 Ekim 2026 · Etiketler: RESMÎ 3. TARAF TAHMİN ÇELİŞKİ

  • SMS göndermek = BTK yetkili işletmecinin abonesi olmak. API anahtarı almak değil, sözleşme imzalamak.
  • Başlık: 3–11 karakter alfanümerik; 1 Nisan 2026'dan beri yalnız rakamlı başlık yok, yurt dışından bağlantılı SMS engelli.
  • Ticari ileti: izin + İYS + iletide kimlik + ret yolu. 2026 cezası 2.859–14.309 TL, toplu gönderimde on katına kadar.
  • Yurt dışı API: Twilio ≈ 0,0305 USD/segment, yalnız OTP/işlemsel, başlık kaydı şart.
  • Alternatif: Telegram Gateway 0,01 USD/kod; WhatsApp authentication şablonu.
  • TurkeySMS v4: bireysel hesap kabul eden, OTP/toplu/rapor/webhook uçlu REST API; gelen SMS, HLR ve ses yok; fiyat sayfası kendi içinde çelişkili.

§ 8C# örnekleri (.NET, HttpClient + System.Text.Json)

Gizli değerler ortam değişkeninden: TURKEYSMS_API_KEY, TURKEYSMS_WEBHOOK_SECRET. Resmî .NET SDK yok, kütüphane gerekmez. İki parça da .NET 10 SDK ile derlendi (uyarısız); webhook parçası yerelde imzalı/imzasız/eski istekle denendi (200 / 403 / 403). API'ye istek atılmadı.

Tekil SMS + OTP + durum sorgusu (konsol)

// Konsol uygulaması, .NET 8+ (ImplicitUsings açık)
using System.Net.Http.Json;
using System.Security.Cryptography;
using System.Text.Json.Nodes;

var apiKey = Environment.GetEnvironmentVariable("TURKEYSMS_API_KEY")
             ?? throw new InvalidOperationException("TURKEYSMS_API_KEY yok");
var http = new HttpClient
{
    BaseAddress = new Uri("https://api.turkeysms.com.tr/"),
    Timeout = TimeSpan.FromSeconds(30)
};

// Ortak çağrı: api_key GÖVDEYE eklenir (başlık yok). Karar result_code'a göre, metne göre değil.
async Task<JsonObject> CallAsync(string path, JsonObject body, CancellationToken ct = default)
{
    body["api_key"] = apiKey;
    for (var attempt = 1; ; attempt++)
    {
        var wait = TimeSpan.FromSeconds(Math.Pow(2, attempt));       // 2, 4, 8 sn
        HttpResponseMessage res;
        try { res = await http.PostAsJsonAsync(path, body, ct); }
        catch (Exception e) when (attempt < 4 && !ct.IsCancellationRequested
                                  && e is HttpRequestException or TaskCanceledException)
        { await Task.Delay(wait, ct); continue; }                     // ağ / zaman aşımı → tekrar (idempotency korur)

        using (res)
        {
            if ((int)res.StatusCode >= 500 && attempt < 4) { await Task.Delay(wait, ct); continue; }

            var json = await res.Content.ReadFromJsonAsync<JsonObject>(ct)
                       ?? throw new InvalidOperationException($"{path}: boş cevap");
            if (json["result"]?.GetValue<bool>() == true) return json;

            var code = json["result_code"]?.GetValue<string>();
            if (code == "TS-1060" && attempt < 4)                     // dakikalık hız sınırı
            { await Task.Delay(TimeSpan.FromSeconds(60), ct); continue; }

            // TS-1031 anahtar · TS-1027 bakiye · TS-1028 başlık · TS-1065 izin → tekrar deneme YOK
            throw new InvalidOperationException($"{path}: {code} {json["result_message"]}");
        }
    }
}

// "+90 5XX XXX XX XX" / "05XX…" / "5XX…" → "905XXXXXXXXX" (sentto/mobile biçimi: + yok)
static string ToMsisdn(string phone)
{
    var d = new string(phone.Where(char.IsAsciiDigit).ToArray());
    if (d.StartsWith('0')) d = "9" + d;                 // 05XX… → 905XX…
    else if (d.Length == 10) d = "90" + d;              // 5XX…  → 905XX…
    return d.Length == 12 && d.StartsWith("905") ? d : throw new ArgumentException("TR cep numarası değil");
}

// 1) Tekil SMS — bilgilendirme (işlemsel)
var sms = await CallAsync("sms/send", new JsonObject
{
    ["title"] = "MARKAADI",                           // onaylı başlık, büyük/küçük harf duyarlı
    ["sentto"] = ToMsisdn("+90 5XX XXX XX XX"),       // yer tutucu; gerçek numarayla değiştir
    ["text"] = "Siparişiniz kargoya verildi. Takip no: {TAKIP_NO}",
    ["sms_lang"] = 1,                                  // 0 EN · 1 TR · 2 AR/Unicode (vars.)
    ["content_type"] = 0                               // 0 işlemsel · 1 yüksek kalite · 2 reklam
});
var smsId = sms["sms_id"]!.GetValue<long>();
Console.WriteLine($"sms_id={smsId} segment={sms["number_of_sms"]}");

// 2) OTP — kodu TurkeySMS üretir; metinde TS-CODE zorunlu. Cevapta kod DÖNMEZ.
await CallAsync("otp/detailed", new JsonObject
{
    ["mobile"] = ToMsisdn("+90 5XX XXX XX XX"),
    ["title"] = "MARKAADI",
    ["text"] = "MARKAADI dogrulama kodunuz: TS-CODE. 5 dakika gecerlidir, kimseyle paylasmayin.",   // Türkçe karaktersiz → 1 segment
    ["lang"] = 1,
    ["digits"] = 6
});

// 2b) Kodu kendin doğrulayacaksan: kendin üret, /sms/send ile işlemsel gönder, özetini sakla
var code = RandomNumberGenerator.GetInt32(0, 1_000_000).ToString("D6");
await CallAsync("sms/send", new JsonObject
{
    ["title"] = "MARKAADI",
    ["sentto"] = ToMsisdn("+90 5XX XXX XX XX"),
    ["text"] = $"MARKAADI dogrulama kodunuz: {code}. 5 dakika gecerlidir, kimseyle paylasmayin.",
    ["sms_lang"] = 1,
    ["content_type"] = 0
});
// sakla: SHA256(code + kullanıcı), son geçerlilik = şimdi + 5 dk, deneme sayacı (ör. en çok 5)

// 3) Durum sorgula (webhook yoksa): queued → sent → delivered | failed | expired
var st = await CallAsync("sms/status", new JsonObject { ["sms_id"] = smsId });
Console.WriteLine(st["status"]);

    var parts = req.Headers["X-TurkeySMS-Signature"].ToString().Split('=', 2);
    if (parts.Length != 2 || parts[0] != "sha256") return Results.StatusCode(403);
    byte[] got;
    try { got = Convert.FromHexString(parts[1]); } catch (FormatException) { return Results.StatusCode(403); }
    if (!CryptographicOperations.FixedTimeEquals(got, HMACSHA256.HashData(secret, raw)))
        return Results.StatusCode(403);                // sabit zamanlı karşılaştırma

    // 3) Tekrar oynatma: X-TurkeySMS-Timestamp ±5 dk (ISO-8601; Unix saniyesi gelirse onu da kabul et)
    var tsText = req.Headers["X-TurkeySMS-Timestamp"].ToString();
    DateTimeOffset ts;
    if (long.TryParse(tsText, out var unix)) ts = DateTimeOffset.FromUnixTimeSeconds(unix);
    else if (!DateTimeOffset.TryParse(tsText, CultureInfo.InvariantCulture,
                                      DateTimeStyles.AssumeUniversal, out ts)) return Results.StatusCode(403);
    if ((DateTimeOffset.UtcNow - ts).Duration() > TimeSpan.FromMinutes(5)) return Results.StatusCode(403);

    // 4) Ayrıştır, tekilleştir, hemen 200 dön (2xx dışı = yeniden deneme)
    var ev = JsonNode.Parse(raw)!;
    var type = ev["event_type"]?.GetValue<string>();      // sms.sent | sms.delivered | sms.failed | otp.verified | key.test
    var smsId = ev["data"]?["sms_id"]?.GetValue<long>();
    var status = ev["data"]?["status"]?.GetValue<string>();
    if (!seen.TryAdd($"{type}:{smsId}:{status}", 0)) return Results.Ok();   // aynı olay ikinci kez

    // Sıra garanti değil: "delivered" kaydedildiyse geç gelen "sent" durumu geri almasın.
    // İşi kuyruğa at (DB güncelle vb.), burada bekletme.
    Console.WriteLine($"{type} sms_id={smsId} status={status}");
    return Results.Ok();
});

app.Run();
